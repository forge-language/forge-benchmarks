cmake_minimum_required(VERSION 3.20)
project(forge_benchmarks LANGUAGES C)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Forge CONFIG REQUIRED)
find_package(ForgeRuntime CONFIG REQUIRED)
find_package(ForgeStd CONFIG REQUIRED)
set(FORGE_ROOT "${FORGE_ROOT_DIR}" CACHE PATH "Installed Forge SDK prefix (bin, include, lib)")
if(NOT FORGE_ROOT)
  message(FATAL_ERROR "Set FORGE_ROOT to the installed Forge SDK prefix")
endif()
find_program(FORGE_EXECUTABLE forge HINTS "${FORGE_ROOT}/bin" REQUIRED NO_DEFAULT_PATH)
foreach(name string scheduler)
  if(name STREQUAL "scheduler" AND WIN32)
    continue()
  endif()
  add_executable(${name}_bench benchmark/${name}_bench.c)
  target_link_libraries(${name}_bench PRIVATE ForgeRuntime::runtime ForgeStd::stdlib)
endforeach()
set(FORGE_RUNTIME_SOURCE_DIR "" CACHE PATH "Runtime checkout for the private queue benchmark")
if(FORGE_RUNTIME_SOURCE_DIR)
  if(NOT EXISTS "${FORGE_RUNTIME_SOURCE_DIR}/src/work_queue.h")
    message(FATAL_ERROR "FORGE_RUNTIME_SOURCE_DIR must contain src/work_queue.h")
  endif()
  add_executable(work_queue_bench benchmark/work_queue_bench.c)
  target_include_directories(work_queue_bench PRIVATE "${FORGE_RUNTIME_SOURCE_DIR}/src")
  target_link_libraries(work_queue_bench PRIVATE ForgeRuntime::runtime)
endif()
if(UNIX)
  add_executable(c_bench_server benchmark/c/bench_server.c)
  find_package(Threads REQUIRED)
  target_link_libraries(c_bench_server PRIVATE Threads::Threads)
endif()
option(FORGE_BUILD_HTTP_BENCHMARKS "Compile HTTP fixtures requiring network library dependencies" ON)
set(fixtures bench_coro)
if(FORGE_BUILD_HTTP_BENCHMARKS)
  list(APPEND fixtures bench_server bench_hybrid_server bench_sendfile_server bench_uring_server bench_tls_server bench_routing_server)
endif()
foreach(name IN LISTS fixtures)
  set(output "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/${name}${CMAKE_EXECUTABLE_SUFFIX}")
  add_custom_command(OUTPUT "${output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}"
    COMMAND "${FORGE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/benchmark/forge/${name}.fg"
      -o "${output}" --cc "${CMAKE_C_COMPILER}" --forge-root "${FORGE_ROOT}" --lib-dir "${FORGE_ROOT}/lib"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/benchmark/forge/${name}.fg" "${FORGE_EXECUTABLE}"
    VERBATIM)
  add_custom_target(${name} DEPENDS "${output}")
endforeach()
add_custom_target(benchmarks DEPENDS ${fixtures})
